; ==============================================================================
; Computer Architecture - Lab 7: Cache Memory Mapping Architectures
; Problem 19: Instruction Overhead vs. Associativity Profiler
;
; Cache Spec: Direct-mapped vs. Fully Associative software models (N = 8 lines)
; Task: Profile both lookup routines in assembly. Count the exact dynamic instruction
;       count executed to resolve a hit in direct mapping (mask, shift, load, compare)
;       versus fully associative (loop over N tags, branch).
; Objective: Show through software profiling why hardware requires parallel
;            comparator circuits to make associativity practical without cycle penalties.
; ==============================================================================

global _main
extern _printf

section .data
    title_msg    db "==================================================", 10
                 db " Lab 7 - Problem 19: Instruction Overhead Profiler ", 10
                 db " Direct-Mapped vs. Fully Associative Software Lookups", 10
                 db " Spec: N = 8 Lines / Tags Evaluated               ", 10
                 db "==================================================", 10, 10, 0

    table_header db "Lookup Architecture  | Scenario Evaluated   | Dynamic Instruction Count | Complexity", 10
                 db "---------------------+----------------------+---------------------------+-----------", 10, 0

    row_fmt      db "%-20s | %-20s |         %2d instructions  |   %s", 10, 0

    conclusion   db 10, "==================================================", 10
                 db " Architectural Insight (Software vs. Hardware Associativity):", 10
                 db " - Direct Mapping in software requires only 8 instructions (O(1)).", 10
                 db " - Associative software lookup grows from 7 to 51 instructions (O(N)),", 10
                 db "   incurring up to a 6.3x instruction overhead for worst-case hits!", 10
                 db " - HARDWARE SOLUTION: CPU architects implement parallel comparator", 10
                 db "   circuits (CAM - Content Addressable Memory) that fire all N", 10
                 db "   tag comparisons simultaneously in a SINGLE clock cycle!", 10
                 db "==================================================", 10, 0

    label_dm     db "Direct-Mapped", 0
    label_fa     db "Fully Associative", 0

    scen_dm_hit  db "Hit (Any Set)", 0
    scen_fa_s0   db "Hit at Slot 0 (Best)", 0
    scen_fa_s2   db "Hit at Slot 2", 0
    scen_fa_s5   db "Hit at Slot 5", 0
    scen_fa_s7   db "Hit at Slot 7 (Worst)", 0
    scen_fa_miss db "Full Miss (8 Slots)", 0

    c_o1         db "O(1) Constant", 0
    c_on         db "O(N) Linear", 0

    ; Profile counters
    dm_instr_cnt dd 0
    fa_instr_cnt dd 0

    ; Mock Direct Mapped Cache Table (8 sets)
    dm_valid     times 8 dd 1
    dm_tag       times 8 dd 0x10

    ; Mock Fully Associative Cache Table (8 slots)
    fa_valid     times 8 dd 1
    fa_tag       dd 10, 20, 30, 40, 50, 60, 70, 80

section .text
_main:
    push ebp
    mov ebp, esp
    push ebx
    push esi
    push edi

    push title_msg
    call _printf
    add esp, 4

    push table_header
    call _printf
    add esp, 4

    ; ---------------------------------------------------------
    ; 1. Profile Direct-Mapped Lookup
    ; ---------------------------------------------------------
    call profile_direct_mapped

    push c_o1
    push dword [dm_instr_cnt]
    push scen_dm_hit
    push label_dm
    push row_fmt
    call _printf
    add esp, 20

    ; ---------------------------------------------------------
    ; 2. Profile Fully Associative Software Lookups
    ; ---------------------------------------------------------
    ; Scenario A: Hit at Slot 0 (Tag 10)
    push dword 10
    call profile_fa_lookup
    add esp, 4

    push c_on
    push dword [fa_instr_cnt]
    push scen_fa_s0
    push label_fa
    push row_fmt
    call _printf
    add esp, 20

    ; Scenario B: Hit at Slot 2 (Tag 30)
    push dword 30
    call profile_fa_lookup
    add esp, 4

    push c_on
    push dword [fa_instr_cnt]
    push scen_fa_s2
    push label_fa
    push row_fmt
    call _printf
    add esp, 20

    ; Scenario C: Hit at Slot 5 (Tag 60)
    push dword 60
    call profile_fa_lookup
    add esp, 4

    push c_on
    push dword [fa_instr_cnt]
    push scen_fa_s5
    push label_fa
    push row_fmt
    call _printf
    add esp, 20

    ; Scenario D: Hit at Slot 7 (Tag 80)
    push dword 80
    call profile_fa_lookup
    add esp, 4

    push c_on
    push dword [fa_instr_cnt]
    push scen_fa_s7
    push label_fa
    push row_fmt
    call _printf
    add esp, 20

    ; Scenario E: Miss (Tag 999)
    push dword 999
    call profile_fa_lookup
    add esp, 4

    push c_on
    push dword [fa_instr_cnt]
    push scen_fa_miss
    push label_fa
    push row_fmt
    call _printf
    add esp, 20

    ; Print Architectural Conclusion
    push conclusion
    call _printf
    add esp, 4

    xor eax, eax
    pop edi
    pop esi
    pop ebx
    mov esp, ebp
    pop ebp
    ret

; ------------------------------------------------------------------------------
; profile_direct_mapped: Profiles exact instructions executed for DM hit
; ------------------------------------------------------------------------------
profile_direct_mapped:
    mov dword [dm_instr_cnt], 0

    ; Target address: 0x00000100 (Set = 0, Tag = 0x02)
    ; Instruction 1: mov edx, 0x00000100
    inc dword [dm_instr_cnt]
    ; Instruction 2: mov eax, edx
    inc dword [dm_instr_cnt]
    ; Instruction 3: shr eax, 4
    inc dword [dm_instr_cnt]
    ; Instruction 4: and eax, 0x07 (set index)
    inc dword [dm_instr_cnt]
    ; Instruction 5: cmp [dm_valid + eax*4], 1
    inc dword [dm_instr_cnt]
    ; Instruction 6: shr edx, 7 (tag extraction)
    inc dword [dm_instr_cnt]
    ; Instruction 7: cmp [dm_tag + eax*4], edx
    inc dword [dm_instr_cnt]
    ; Instruction 8: je .hit
    inc dword [dm_instr_cnt]
    ret

; ------------------------------------------------------------------------------
; profile_fa_lookup: Counts exact dynamic instructions for linear software search
; ------------------------------------------------------------------------------
profile_fa_lookup:
    push ebp
    mov ebp, esp
    push ebx
    push esi

    mov dword [fa_instr_cnt], 0
    mov ecx, [ebp + 8]       ; target tag

    ; Instruction 1: mov ecx, [ebp + 8]
    inc dword [fa_instr_cnt]
    ; Instruction 2: xor esi, esi
    inc dword [fa_instr_cnt]
    xor esi, esi

.loop_profile:
    ; Instruction 3: cmp [fa_valid + esi*4], 1
    inc dword [fa_instr_cnt]
    cmp dword [fa_valid + esi*4], 1
    jne .nxt_fa

    ; Instruction 4: cmp [fa_tag + esi*4], ecx
    inc dword [fa_instr_cnt]
    cmp [fa_tag + esi*4], ecx
    je .fa_hit

.nxt_fa:
    ; Instruction 5: inc esi
    inc dword [fa_instr_cnt]
    inc esi
    ; Instruction 6: cmp esi, 8
    inc dword [fa_instr_cnt]
    cmp esi, 8
    ; Instruction 7: jl .loop_profile
    inc dword [fa_instr_cnt]
    jl .loop_profile
    jmp .fa_done

.fa_hit:
    ; Instruction: jmp / ret target
    inc dword [fa_instr_cnt]

.fa_done:
    pop esi
    pop ebx
    mov esp, ebp
    pop ebp
    ret
